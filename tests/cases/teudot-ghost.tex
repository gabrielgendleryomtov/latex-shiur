% source has the same lines; only the colours differ.

\usepackage{shiur}
\usepackage{teudot}
\setshiurfont{hebrew}{FreeSerif}
\declareteudah{e}{red}
\declareteudah{j}{blue}
\begin{document}
\begin{mekor}[\tanach{exodus}{14:19-20}]
\teudotsetup{show=j, others=ghost}
\begin{mekorlang}{english}

\end{mekorlang}
\end{mekor}
\begin{mekor}[\tanach{exodus}{14:19-20}]

\end{mekor}
\end{document}
